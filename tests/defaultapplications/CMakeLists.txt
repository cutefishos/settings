cmake_minimum_required(VERSION 3.22)
project(defaultapplications_test LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
find_package(CutefishFramework REQUIRED COMPONENTS Application)
add_executable(tst_mimeapps tst_mimeapps.cpp ../../src/defaultapplications.cpp)
target_include_directories(tst_mimeapps PRIVATE ../../src)
target_link_libraries(tst_mimeapps PRIVATE Qt6::Core Qt6::Test Cutefish::Application)
enable_testing()
add_test(NAME mimeapps COMMAND tst_mimeapps)

cmake_minimum_required(VERSION 3.22)
project(cursor_theme_test LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui DBus Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(Xcursor REQUIRED IMPORTED_TARGET xcursor)
find_package(CutefishFramework REQUIRED COMPONENTS Application)
find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
add_executable(cursor_theme_test cursor_theme_test.cpp ../../src/cursor/cursorthememodel.cpp
    ../../src/cursor/cursortheme.cpp)
target_include_directories(cursor_theme_test PRIVATE ../../src)
target_link_libraries(cursor_theme_test PRIVATE Qt6::Core Qt6::Gui Qt6::DBus Qt6::Test PkgConfig::Xcursor
    Cutefish::Application)
enable_testing()
add_test(NAME cursor_theme_test COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:cursor_theme_test>)
set_tests_properties(cursor_theme_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
